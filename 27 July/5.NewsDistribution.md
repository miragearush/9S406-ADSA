/**

#include <bits/stdc++.h>
using namespace std;

class DisjointSet {
    
public:
    vector<int>rank,parent,size;
    DisjointSet(int n) {
        rank.resize(n+1,0);
        parent.resize(n+1);
        size.resize(n+1,1);
        for(int i=0;i<=n;i++) {
            parent[i]=i;
        }
    }
    
    int findUParent(int u) {
        if(parent[u]==u)
            return u;
        return parent[u]=findUParent(parent[u]);
    }
    void unionByRank(int u,int v) {
        int ulp_u=findUParent(u);
        int ulp_v=findUParent(v);
        
        if(ulp_u==ulp_v)
            return;
        if(rank[ulp_u]<rank[ulp_v]) {
            parent[ulp_u]=ulp_v;
        }
        else if(rank[ulp_v]<rank[ulp_u]) {
            parent[ulp_v]=ulp_u;
        }
        else {
            parent[ulp_v]=ulp_u;
            rank[ulp_u]++;
        }
    }
    void unionBySize(int u,int v) {
        int ulp_u=findUParent(u);
        int ulp_v=findUParent(v);
        
        if(ulp_u==ulp_v)
            return;
        if(size[ulp_u]<size[ulp_v]) {
            parent[ulp_u]=ulp_v;
            size[ulp_v]+=size[ulp_u];
        }
        else {
            parent[ulp_v]=ulp_u;
            size[ulp_u]+=size[ulp_v];
        }

    }
};

int main() 
{
    int n,m;
    cin>>n>>m;

    DisjointSet dsu(n);

    for(int i=0;i<m;i++) {
        int sz;
        cin>>sz;
        for(int j=0;j<sz;j++) {
            int u,v;
            if(j==0)
                cin>>u;
            else {
                cin>>v;
                dsu.unionBySize(u,v);
            }
        }
    }
    for(int i=1;i<=n;i++) {
        int ulp_i=dsu.findUParent(i);
        cout<<dsu.size[ulp_i]<<" ";
    }
    return 0;
}
